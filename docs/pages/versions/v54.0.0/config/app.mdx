// App.js
import React, { useState } from 'react';
import { View, Text, FlatList, Image, TouchableOpacity, StyleSheet, Alert } from 'react-native';

const products = [
  { id: '1', name: "Abhishek-ka-juta", price: 999, image: "https://share.google/images/niMNaiTYjUBrj4S6V" },
  { id: '2', name: "Ankit-ka-juta", price: 1999, image: "https://share.google/images/qtlZ9ZhMTI1ESRNCM" },
  { id: '3', name: "Saras-ka-juta", price: 499, image: "https://share.google/images/hTutcO1GBBCQUs0P8" },
];

export default function App() {
  const [cart, setCart] = useState([]);

  const addToCart = (item) => {
    setCart([...cart, item]);
    Alert.alert("Added to Cart", `${item.name} added successfully!`);
  };

  const renderItem = ({ item }) => (
    <View style={styles.card}>
      <Image source={{ uri: item.image }} style={styles.img} />
      <Text style={styles.title}>{item.name}</Text>
      <Text style={styles.price}>₹{item.price}</Text>
      <TouchableOpacity style={styles.btn} onPress={() => addToCart(item)}>
        <Text style={{ color: "#fff", fontWeight: "600" }}>Add to Cart</Text>
      </TouchableOpacity>
    </View>
  );

  return (
    <View style={styles.container}>
      <Text style={styles.header}>🛒 Antony Store</Text>
      <FlatList
        data={products}
        renderItem={renderItem}
        keyExtractor={(item) => item.id}
      />
      <View style={styles.cartBox}>
        <Text style={{ fontWeight: "700" }}>Cart Items: {cart.length}</Text>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: "#f2f2f2" },
  header: { fontSize: 22, fontWeight: "bold", textAlign: "center", margin: 12 },
  card: { backgroundColor: "#fff", margin: 12, padding: 12, borderRadius: 8, elevation: 3 },
  img: { width: "100%", height: 180, borderRadius: 6 },
  title: { fontSize: 18, fontWeight: "600", marginTop: 6 },
  price: { fontSize: 16, color: "green", marginVertical: 4 },
  btn: { backgroundColor: "#007bff", padding: 10, borderRadius: 6, alignItems: "center" },
  cartBox: { backgroundColor: "#fff", padding: 12, alignItems: "center" }
});
